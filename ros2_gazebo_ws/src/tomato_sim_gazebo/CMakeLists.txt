cmake_minimum_required(VERSION 3.8)
project(tomato_sim_gazebo)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)

add_library(tomato_sim_joint_initializer SHARED src/initial_joint_state_plugin.cpp)
target_include_directories(tomato_sim_joint_initializer PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(tomato_sim_joint_initializer ${GAZEBO_LIBRARIES})
target_compile_options(tomato_sim_joint_initializer PRIVATE ${GAZEBO_CXX_FLAGS})
install(TARGETS tomato_sim_joint_initializer LIBRARY DESTINATION lib)
add_library(tomato_sim_gripper_collision_filter SHARED src/gripper_collision_filter.cpp)
target_include_directories(tomato_sim_gripper_collision_filter PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(tomato_sim_gripper_collision_filter ${GAZEBO_LIBRARIES})
target_compile_options(tomato_sim_gripper_collision_filter PRIVATE ${GAZEBO_CXX_FLAGS})
install(TARGETS tomato_sim_gripper_collision_filter LIBRARY DESTINATION lib)
add_library(tomato_sim_slide_parking_brake SHARED src/slide_parking_brake.cpp)
target_include_directories(tomato_sim_slide_parking_brake PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(tomato_sim_slide_parking_brake ${GAZEBO_LIBRARIES})
target_compile_options(tomato_sim_slide_parking_brake PRIVATE ${GAZEBO_CXX_FLAGS})
ament_target_dependencies(tomato_sim_slide_parking_brake gazebo_ros rclcpp std_msgs)
install(TARGETS tomato_sim_slide_parking_brake LIBRARY DESTINATION lib)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_python_install_package(tomato_sim_config)
ament_environment_hooks(environment/workspace.sh.in)
ament_package()
